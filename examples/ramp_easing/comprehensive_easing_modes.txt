# ===================================================================================
# COMPREHENSIVE EASING MODES EXAMPLE (v2.2.3)
# ===================================================================================
# This file demonstrates ALL available easing modes and protocol options
# Use with: python mock_arduino.py examples/comprehensive_easing_modes.txt --plot
# ===================================================================================

# ============================================================================
# SECTION 1: BASIC EASING MODES (L, C, I, O)
# ============================================================================
# All modes use format: (MODE:start_pwm,end_pwm,duration_ms)
# 
# L = Linear      - Constant speed, uniform rate of change
# C = Cosine      - Smooth S-curve, ease-in-out (slow→fast→slow)
# I = Ease-In     - Slow start, accelerates toward end
# O = Ease-Out    - Fast start, decelerates toward end

# Channel 1: Linear fade up
PATTERN:1;CH:1;RAMP:(L:0,255,5000);REPEATS:2

# Channel 2: Cosine smooth S-curve
PATTERN:1;CH:2;RAMP:(C:0,255,5000);REPEATS:2

# Channel 3: Ease-In (slow start)
PATTERN:1;CH:3;RAMP:(I:0,255,5000);REPEATS:2

# Channel 4: Ease-Out (slow end)
PATTERN:1;CH:4;RAMP:(O:0,255,5000);REPEATS:2

# ============================================================================
# SECTION 2: X MODE - CUSTOM COSINE CURVE RANGES
# ============================================================================
# X mode uses: (X:start_pwm,end_pwm,duration_ms|t_start,t_end)
# 
# The t parameter controls which part of the cosine curve to use:
#   t=0→1: Rising phase (0% → 100%)
#   t=1→2: Falling phase (100% → 0%)
#   t=0→2: Full breathing cycle (0% → 100% → 0%)
#   t=0→0.5: Partial rise (start of rise, slow)
#   t=0.5→1: Partial rise (end of rise, fast)
#
# Note: start_pwm and end_pwm are NOT needed in X mode - PWM = 255 * f(t)

# Channel 5: X mode breathing (full cycle 0→2)
PATTERN:1;CH:5;RAMP:(X:4000|0,2);REPEATS:3

# ============================================================================
# SECTION 3: F MODE - CUSTOM FUNCTIONS
# ============================================================================
# F mode uses: (F:function_name,duration_ms)
# 
# Only 2 parameters: function name and duration
# The function controls the entire curve shape from progress=[0,1]
#
# Available built-in functions:
#   sine_wave    - Half sine: 0→255→0
#   double_sine  - Two bumps: 0→255→0→255→0
#   bounce       - Damped oscillation (ball bounce)
#   heartbeat    - Lub-dub cardiac pattern
#   exp_decay    - Sharp start, slow fade
#   log_rise     - Slow start, accelerating
#   step_50      - Jumps to full at 50% progress
#   sawtooth     - 3 linear cycles
#   breathing    - Smooth cosine in-out
#   flicker      - Irregular oscillation pattern

# Channel 6: Heartbeat effect
PATTERN:1;CH:6;RAMP:(F:heartbeat,2000);REPEATS:5

# Channel 7: Bounce effect
PATTERN:1;CH:7;RAMP:(F:bounce,3000);REPEATS:3

# Channel 8: Breathing + flicker sequence
PATTERN:1;CH:8;RAMP:(F:breathing,4000),(F:flicker,2000);REPEATS:2

# ============================================================================
# SECTION 4: MULTI-SEGMENT RAMPS
# ============================================================================
# Combine multiple segments in a single RAMP command
# Segments execute in sequence

# Channel 1: Pattern 2 - Fade up + hold + fade down
PATTERN:2;CH:1;RAMP:(I:0,255,3000),(L:255,255,2000),(O:255,0,3000);REPEATS:1

# Channel 2: Pattern 2 - Complex transition
PATTERN:2;CH:2;RAMP:(L:0,128,1000),(C:128,255,2000),(L:255,255,1000),(O:255,0,2000);REPEATS:1

# ============================================================================
# SECTION 5: PWM STATUS PATTERNS (NON-RAMP)
# ============================================================================
# Traditional STATUS/TIME_MS format with PWM intensity values (0-255)

# Channel 3: Step brightness levels
PATTERN:2;CH:3;STATUS:0,64,128,191,255;TIME_MS:1000,1000,1000,1000,1000;REPEATS:2

# Channel 4: Binary blink at full brightness
PATTERN:2;CH:4;STATUS:255,0;TIME_MS:500,500;REPEATS:10

# ============================================================================
# SECTION 6: PULSE MODULATION
# ============================================================================
# PULSE format: T<period_ms>pw<pulse_width_ms>
# Creates rapid on-off switching within each time segment

# Channel 5: Pattern 2 - Pulsing at 2Hz (T500pw250 = 50% duty cycle)
PATTERN:2;CH:5;STATUS:255,0;TIME_MS:5000,2000;REPEATS:2;PULSE:T500pw250,

# ============================================================================
# SECTION 7: MIXED MODES IN SEQUENCE
# ============================================================================
# Combine F mode with standard modes in multi-segment ramps

# Channel 6: Pattern 2 - Heartbeat intro → linear fade → bounce outro
PATTERN:2;CH:6;RAMP:(F:heartbeat,3000),(L:255,128,2000),(F:bounce,3000);REPEATS:2

# Channel 7: Pattern 2 - X mode transition + F mode effect
PATTERN:2;CH:7;RAMP:(X:0,255,2000|0,1),(F:flicker,1500),(X:0,255,2000|1,2);REPEATS:2

# ============================================================================
# START_TIME (REQUIRED)
# ============================================================================
# All 8 channels start at the same time
START_TIME: {'CH1': 0, 'CH2': 0, 'CH3': 0, 'CH4': 0, 'CH5': 0, 'CH6': 0, 'CH7': 0, 'CH8': 0}

# ============================================================================
# WAIT_STATUS (OPTIONAL)
# ============================================================================
# PWM intensity during countdown (0.0-1.0)
WAIT_STATUS: {'CH1': 0, 'CH2': 0, 'CH3': 0.1, 'CH4': 0, 'CH5': 0, 'CH6': 0.05, 'CH7': 0, 'CH8': 0}

# ============================================================================
# CALIBRATION_FACTOR (OPTIONAL)
# ============================================================================
CALIBRATION_FACTOR: 1.0

# ===================================================================================
# QUICK REFERENCE
# ===================================================================================
#
# RAMP MODES:
#   L - Linear:    (L:0,255,5000)           Constant speed
#   C - Cosine:    (C:0,255,5000)           S-curve (ease-in-out)
#   I - Ease-In:   (I:0,255,5000)           Slow start
#   O - Ease-Out:  (O:0,255,5000)           Slow end
#   X - Custom t:  (X:0,255,5000|0,2)       Raw cosine curve range
#   F - Function:  (F:heartbeat,2000)       Custom function
#
# SEGMENT FORMAT:
#   Standard:  (MODE:start,end,duration)
#   X mode:    (X:start,end,duration|t_start,t_end)
#   F mode:    (F:func_name,duration)        # No start/end PWM!
#
# MULTI-SEGMENT:
#   RAMP:(seg1),(seg2),(seg3);
#
# PULSE FORMAT:
#   PULSE:T<period>pw<width>,T<period>pw<width>,...
#   Example: T1000pw500 = 1 second period, 500ms on time (50% duty)
#
# ===================================================================================
